// src/config/database.js
// Configuração da ligação ao PostgreSQL para a Cristal Water

const { Pool } = require("pg");

const pool = new Pool({
  host: process.env.DB_HOST || "localhost",
  port: process.env.DB_PORT ? Number(process.env.DB_PORT) : 5432,
  user: process.env.DB_USER || "postgres",
  password: process.env.DB_PASSWORD || "",
  database: process.env.DB_DATABASE || "cristalwater",
});

pool.on("error", (err) => {
  console.error("Erro inesperado na ligação à base de dados:", err);
});

async function testConnection() {
  try {
    const result = await pool.query("SELECT NOW() AS now");
    console.log(
      "[DB] Ligação PostgreSQL OK - Hora do servidor:",
      result.rows[0].now
    );
  } catch (err) {
    console.error("[DB] Erro ao testar ligação à base de dados:", err.message);
  }
}

testConnection();

module.exports = {
  pool,
};
